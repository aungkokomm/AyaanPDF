            <ScrollView
                x:Name="PageScroller"
                Grid.Row="2"
                ZoomMode="Enabled"
                HorizontalScrollBarVisibility="Auto"
                VerticalScrollBarVisibility="Auto"
                HorizontalScrollMode="Enabled"
                VerticalScrollMode="Enabled"
                MinZoomFactor="0.1"
                MaxZoomFactor="8"
                ViewChanged="PageScroller_ViewChanged"
                SizeChanged="PageScroller_SizeChanged">
                <!--
                    Width/Height are pinned to the ViewModel's layout size in
                    DIPs and the Image stretches to fill it. Sharper renders
                    therefore change only pixel density, never layout, so
                    ScrollView.ZoomFactor 1.0 reliably means "fits the width".
                -->
                <controls:CursorGrid
                    x:Name="ViewportHost"
                    Background="Transparent"
                    HorizontalAlignment="Center"
                    VerticalAlignment="Top"
                    Padding="24"
                    PointerPressed="ViewportHost_PointerPressed"
                    PointerMoved="ViewportHost_PointerMoved"
                    PointerReleased="ViewportHost_PointerReleased"
                    PointerCanceled="ViewportHost_PointerReleased"
                    PointerCaptureLost="ViewportHost_PointerReleased">

                    <!--
                        Continuous page stack.

                        Each card is sized in SLOT space (unzoomed DIPs) from
                        the page's intrinsic size, fixed the moment the
                        document opens. Zoom is the ScrollView's own zoom
                        factor, which grows the scrollable extent, so a card
                        always occupies its final space and bitmaps can stream
                        in or be released without anything below shifting.

                        ItemsControl, not ItemsRepeater: the slots already
                        virtualize their PIXELS in the view model (bitmaps are
                        released outside the window while heights stay), so
                        virtualizing the containers as well would buy little
                        and would make scroll-offset-to-page mapping depend on
                        realization order.
                    -->
                    <ItemsControl ItemsSource="{x:Bind ViewModel.PageSlots}">
                        <ItemsControl.ItemsPanel>
                            <ItemsPanelTemplate>
                                <StackPanel Orientation="Vertical" Spacing="16" HorizontalAlignment="Center" />
                            </ItemsPanelTemplate>
                        </ItemsControl.ItemsPanel>
                        <ItemsControl.ItemTemplate>
                            <DataTemplate x:DataType="viewmodels:PageSlot">
                                <!-- The page card: white sheet, thin border, drop shadow. -->
                                <Border Width="{x:Bind SlotWidth}" Height="{x:Bind SlotHeight}"
                                        Background="White"
                                        BorderBrush="{ThemeResource ControlStrokeColorDefaultBrush}"
                                        BorderThickness="1"
                                        HorizontalAlignment="Center">
                                    <Border.Shadow>
                                        <ThemeShadow />
                                    </Border.Shadow>
                                    <Grid>
                                        <Image Source="{x:Bind Bitmap, Mode=OneWay}" Stretch="Fill" />

                                        <!--
                                            Overlay layer for this page only.
                                            Marks are stored NORMALIZED, so the
                                            scale is the card's slot width and a
                                            re-render at any resolution cannot
                                            move them.
                                        -->
                                        <Grid HorizontalAlignment="Left" VerticalAlignment="Top">
                                            <Grid.RenderTransform>
                                                <ScaleTransform ScaleX="{x:Bind OverlayScale}" ScaleY="{x:Bind OverlayScale}" />
                                            </Grid.RenderTransform>

                                            <ItemsControl ItemsSource="{x:Bind SearchMatchRects}" IsHitTestVisible="False">
                                                <ItemsControl.ItemsPanel>
                                                    <ItemsPanelTemplate><Grid /></ItemsPanelTemplate>
                                                </ItemsControl.ItemsPanel>
                                                <ItemsControl.ItemTemplate>
                                                    <DataTemplate x:DataType="viewport:TextRect">
                                                        <Rectangle Fill="#66FFA500" Width="{x:Bind Width}" Height="{x:Bind Height}"
                                                                   Margin="{x:Bind local:MainPage.Offset(Left, Top)}"
                                                                   HorizontalAlignment="Left" VerticalAlignment="Top" />
                                                    </DataTemplate>
                                                </ItemsControl.ItemTemplate>
                                            </ItemsControl>

                                            <ItemsControl ItemsSource="{x:Bind SelectionRects}" IsHitTestVisible="False">
                                                <ItemsControl.ItemsPanel>
                                                    <ItemsPanelTemplate><Grid /></ItemsPanelTemplate>
                                                </ItemsControl.ItemsPanel>
                                                <ItemsControl.ItemTemplate>
                                                    <DataTemplate x:DataType="viewport:TextRect">
                                                        <Rectangle Fill="#665599FF" Width="{x:Bind Width}" Height="{x:Bind Height}"
                                                                   Margin="{x:Bind local:MainPage.Offset(Left, Top)}"
                                                                   HorizontalAlignment="Left" VerticalAlignment="Top" />

                                                </ItemsControl.ItemTemplate>
                                            </ItemsControl>

                                            <ItemsControl ItemsSource="{x:Bind Highlights}" IsHitTestVisible="False">
                                                <ItemsControl.ItemsPanel>
                                                    <ItemsPanelTemplate><Grid /></ItemsPanelTemplate>
                                                </ItemsControl.ItemsPanel>
                                                <ItemsControl.ItemTemplate>
                                                    <DataTemplate x:DataType="viewport:HighlightAnnotation">
                                                        <ItemsControl ItemsSource="{x:Bind Rects}">
                                                            <ItemsControl.ItemsPanel>
                                                                <ItemsPanelTemplate><Grid /></ItemsPanelTemplate>
                                                            </ItemsControl.ItemsPanel>
                                                            <ItemsControl.ItemTemplate>
                                                                <DataTemplate x:DataType="viewport:TextRect">
                                                                    <Rectangle Fill="#88FFFF00" Width="{x:Bind Width}" Height="{x:Bind Height}"
                                                                               Margin="{x:Bind local:MainPage.Offset(Left, Top)}"
                                                                               HorizontalAlignment="Left" VerticalAlignment="Top" />
                                                                </DataTemplate>
                                                            </ItemsControl.ItemTemplate>
                                                        </ItemsControl>
                                                    </DataTemplate>
                                                </ItemsControl.ItemTemplate>

                    <!--
                        Ink layer, spanning the WHOLE stack rather than sitting
                        inside a page card. Polyline points cannot be bound
                        through x:Bind, so strokes are built in code-behind,
